* Evil
** Evil mode
=evil-want-keybinding= must be nil before evil loads, or evil-collection will complain.
#+begin_src emacs-lisp
  (use-package evil
    :init
    (setq evil-want-integration t
          evil-want-keybinding nil
          evil-search-module 'evil-search
          evil-want-C-u-scroll t
          evil-undo-system 'undo-redo
          evil-shift-width 4
          evil-ex-search-case 'smart)
    :config
    (evil-mode 1))
#+end_src

** Resize mode (smart-splits)
=SPC w r=, then =h= / =j= / =k= / =l= keep resizing until any other key.
#+begin_src emacs-lisp
  (defvar-keymap my/window-resize-map
    :repeat t
    "h" #'shrink-window-horizontally
    "l" #'enlarge-window-horizontally
    "j" #'enlarge-window
    "k" #'shrink-window)

  (defun my/window-resize ()
    "Start resizing the window; h/j/k/l repeat."
    (interactive)
    (set-transient-map my/window-resize-map t)
    (message "Resize: h/j/k/l, any other key to stop"))
#+end_src

* Finding things
The Telescope replacement: vertico draws the list, orderless does the fuzzy
matching, marginalia adds annotations and consult provides the pickers with
live preview.

** Vertico
=M-j= / =M-k= move through results, like the Telescope mappings.
#+begin_src emacs-lisp
  (use-package vertico
    :custom
    (vertico-cycle t)
    :bind (:map vertico-map
                ("M-j" . vertico-next)
                ("M-k" . vertico-previous))
    :config
    (vertico-mode 1))

  ;; Remember minibuffer history, so recent picks come first
  (savehist-mode 1)
#+end_src

** Consult
=F1= lists the project's (git) files, falling back to every file under the
current directory outside a project.
#+begin_src emacs-lisp
  (use-package consult
    :commands (consult-fd consult-ripgrep consult-buffer consult-line consult-imenu))

  (defun my/find-project-files ()
    "Find files in the current project, or under `default-directory'."
    (interactive)
    (if (project-current)
        (project-find-file)
      (consult-fd default-directory)))
#+end_src

** Projects
=SPC f p= switches between projects. Every git repo directly under =~/code= is
picked up on each switch, so new clones show up without visiting them first.
#+begin_src emacs-lisp
  (defun my/switch-project ()
    "Switch project, after registering the repos under ~/code."
    (interactive)
    (let ((inhibit-message t))
      (project-remember-projects-under "~/code/"))
    (call-interactively #'project-switch-project))
#+end_src

** General
Leader key definer: =SPC= in normal/visual/motion state, =M-SPC= everywhere
else (insert state, terminals, ...). The =override= keymap makes the leader
win over mode maps such as dired and magit.
#+begin_src emacs-lisp
  (use-package general
    :after evil
    :config
    (general-create-definer my/leader
      :states '(normal visual motion insert emacs)
      :keymaps 'override
      :prefix "SPC"
      :non-normal-prefix "M-SPC"))
#+end_src

** Basemap
Function keys and escape, same as nixvim's basemap and Telescope keys.

=F5= opens magit. =lazydocker= and =lazysql= run full-window in a ghostel buffer.
The buffer is killed when the program exits, and pressing the key again while
it is running just switches back to it.
#+begin_src emacs-lisp
  (defun my/run-in-terminal (program)
    "Run PROGRAM in its own ghostel buffer, reusing it while it is alive."
    (let ((name (format "*%s*" program)))
      (if (get-buffer name)
          (switch-to-buffer name)
        (require 'ghostel)
        (let ((buf (get-buffer-create name)))
          (switch-to-buffer buf)
          (ghostel-exec buf program)))))

  (defun my/lazydocker () (interactive) (my/run-in-terminal "lazydocker"))
  (defun my/lazysql () (interactive) (my/run-in-terminal "lazysql"))

  (defun my/escape ()
    "Clear search highlights, then do the usual evil escape."
    (interactive)
    (evil-ex-nohighlight)
    (evil-force-normal-state))

  (general-define-key
   :states 'normal
   "<escape>" 'my/escape
   "<f1>" 'my/find-project-files
   "<S-f1>" 'consult-fd
   "<f2>" 'consult-ripgrep
   "<f3>" 'consult-buffer
   "<f4>" 'kill-current-buffer
   "<f5>" 'magit-status
   "<f6>" 'my/lazydocker
   "<f7>" 'my/lazysql)

  (general-define-key
   :states '(normal visual)
   "s" 'evil-avy-goto-char-timer)

  ;; F1 does nothing outside normal state (instead of opening Emacs help)
  (general-define-key
   :states '(insert visual replace operator)
   "<f1>" 'ignore)
#+end_src
