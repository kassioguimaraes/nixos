** Eglot
Starts automatically for every language below. Most servers are eglot's
defaults; the rest are listed explicitly.

vue-language-server 3 only works as a plugin of typescript-language-server,
so =.vue= files run that with the plugin (path set by =emacs.nix=). Script
blocks get full TypeScript support; templates don't.
#+begin_src emacs-lisp
  (defvar my/vue-typescript-plugin nil
    "Path to @vue/typescript-plugin, set from emacs.nix.")

  (defun my/vue-eglot-contact (&rest _)
    "typescript-language-server with the Vue plugin."
    `("typescript-language-server" "--stdio"
      :initializationOptions
      (:plugins [(:name "@vue/typescript-plugin"
                  :location ,my/vue-typescript-plugin
                  :languages ["vue"])])))

  (use-package eglot
    :ensure nil
    :hook ((rust-ts-mode go-ts-mode python-ts-mode
            typescript-ts-mode tsx-ts-mode js-ts-mode
            vue-mode svelte-mode html-ts-mode mhtml-ts-mode mhtml-mode
            php-ts-mode nix-ts-mode json-ts-mode yaml-ts-mode)
           . eglot-ensure)
    :commands (eglot-rename eglot-code-actions eglot-format-buffer)
    :custom
    (eglot-autoshutdown t)
    (eglot-events-buffer-config '(:size 0 :format short))
    (eglot-max-file-watches 5000)
    ;; Connect in the background instead of blocking while the server starts
    (eglot-sync-connect nil)
    (eglot-report-progress nil)
    ;; No lightbulb/margin hints for available code actions (SPC g a lists them)
    (eglot-code-action-indications nil)
    :init
    ;; Servers send large JSON payloads; read them in bigger chunks
    (setq read-process-output-max (* 1024 1024))
    :config
    (dolist (server `((nix-ts-mode . ("nixd"))
                      (php-ts-mode . ("intelephense" "--stdio"))
                      (python-ts-mode . ("pyright-langserver" "--stdio"))
                      (svelte-mode . ("svelteserver" "--stdio"))
                      ((html-ts-mode mhtml-ts-mode mhtml-mode)
                       . ("vscode-html-language-server" "--stdio"))
                      (vue-mode . my/vue-eglot-contact)))
      (add-to-list 'eglot-server-programs server)))
#+end_src

golangci-lint on top of gopls. Eglot replaces all flymake backends in the
buffers it manages, so this one is added back after eglot starts. It lints
the saved package, so results refresh on save.
#+begin_src emacs-lisp
  (defvar-local my/golangci--proc nil)

  (defun my/golangci-flymake (report-fn &rest _)
    "Flymake backend running golangci-lint on the current file's package."
    (when (process-live-p my/golangci--proc)
      (kill-process my/golangci--proc))
    (let* ((source (current-buffer))
           (file (buffer-file-name))
           (default-directory (file-name-directory file)))
      (setq my/golangci--proc
            (make-process
             :name "golangci-lint" :noquery t :connection-type 'pipe
             :buffer (generate-new-buffer " *golangci-lint*")
             :command '("golangci-lint" "run" "--path-mode=abs"
                        "--output.text.path=stdout"
                        "--output.text.print-issued-lines=false"
                        "--output.text.colors=false"
                        "--show-stats=false" "--issues-exit-code=0" "./")
             :sentinel
             (lambda (proc _event)
               (when (memq (process-status proc) '(exit signal))
                 (unwind-protect
                     (when (and (buffer-live-p source)
                                (eq proc (buffer-local-value 'my/golangci--proc source)))
                       (with-current-buffer (process-buffer proc)
                         (goto-char (point-min))
                         (let (diags)
                           (while (re-search-forward
                                   "^\\([^:\n]+\\):\\([0-9]+\\):\\(?:\\([0-9]+\\):\\)? \\(.*\\)$"
                                   nil t)
                             ;; Compile errors come with relative paths and a
                             ;; ": # pkg" header line; skip the header
                             (when (and (file-equal-p (expand-file-name (match-string 1)) file)
                                        (not (string-prefix-p ": " (match-string 4))))
                               (let ((region (flymake-diag-region
                                              source
                                              (string-to-number (match-string 2))
                                              (and (match-string 3)
                                                   (string-to-number (match-string 3))))))
                                 (push (flymake-make-diagnostic
                                        source (car region) (cdr region)
                                        :warning (match-string 4))
                                       diags))))
                           (funcall report-fn diags))))
                   (kill-buffer (process-buffer proc)))))))))

  (defun my/golangci-setup ()
    (when (derived-mode-p 'go-ts-mode)
      (add-hook 'flymake-diagnostic-functions #'my/golangci-flymake nil t)))
  (add-hook 'eglot-managed-mode-hook #'my/golangci-setup)
#+end_src

** Formatting
=SPC f f= formats like none-ls did: nixfmt for Nix, black for Python, the
language server for everything else, then rustywind sorts Tailwind classes in
web files. Nothing formats on save.
#+begin_src emacs-lisp
  (use-package apheleia
    :commands apheleia-format-buffer
    :config
    (add-to-list 'apheleia-formatters '(rustywind "rustywind" "--stdin")))

  (defvar my/formatters
    '((nix-ts-mode . nixfmt)
      (python-ts-mode . black))
    "Major modes formatted by an external formatter instead of the LSP.")

  (defvar my/tailwind-modes
    '(typescript-ts-mode tsx-ts-mode js-ts-mode vue-mode svelte-mode
      html-ts-mode mhtml-ts-mode mhtml-mode)
    "Major modes where rustywind runs after the LSP formats.")

  (defun my/format-buffer ()
    "Format the buffer with its formatter, or the language server."
    (interactive)
    (if-let* ((formatter (alist-get major-mode my/formatters)))
        (apheleia-format-buffer formatter)
      (if (and (fboundp 'eglot-managed-p) (eglot-managed-p)
               (eglot-server-capable :documentFormattingProvider))
          (eglot-format-buffer)
        (message "No formatter for %s" major-mode))
      (when (memq major-mode my/tailwind-modes)
        (apheleia-format-buffer 'rustywind))))
#+end_src

=F5= opens magit. =lazydocker= and =lazysql= run full-window in a ghostel buffer.
The buffer is killed when the program exits, and pressing the key again while
it is running just switches back to it.
#+begin_src emacs-lisp
  (defun my/run-in-terminal (program)
    "Run PROGRAM in its own ghostel buffer, reusing it while it is alive."
    (let ((name (format "*%s*" program)))
      (if (get-buffer name)
          (switch-to-buffer name)
        (require 'ghostel)
        (let ((buf (get-buffer-create name)))
          (switch-to-buffer buf)
          (ghostel-exec buf program)))))

  (defun my/lazydocker () (interactive) (my/run-in-terminal "lazydocker"))
  (defun my/lazysql () (interactive) (my/run-in-terminal "lazysql"))

  (defun my/escape ()
    "Clear search highlights, then do the usual evil escape."
    (interactive)
    (evil-ex-nohighlight)
    (evil-force-normal-state))

  (general-define-key
   :states 'normal
   "<escape>" 'my/escape

   "<S-f1>" 'consult-fd

** Theme fix: gnus face cycle
Emacs 31 makes =gnus-group-news-N= inherit =gnus-group-news-N-empty=, while
base16-theme (used by Stylix) makes the =-empty= faces inherit the other way.
When something loads gnus before the theme, startup fails with "Face
inheritance results in inheritance cycle". Drop the Emacs-side inheritance
before gnus defines these faces; the theme still styles them.
#+begin_src emacs-lisp
  (dolist (face '(gnus-group-mail-1 gnus-group-mail-2 gnus-group-mail-3
                  gnus-group-mail-low
                  gnus-group-news-1 gnus-group-news-2 gnus-group-news-3
                  gnus-group-news-4 gnus-group-news-5 gnus-group-news-6
                  gnus-group-news-low))
    (face-spec-set face '((t :inherit nil)) 'face-defface-spec))
#+end_src
